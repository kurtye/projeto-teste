import {logger} from "firebase-functions";
import {onDocumentCreated} from "firebase-functions/v2/firestore";
import * as admin from "firebase-admin";
import {getWeek, getYear, getWeekYear} from "date-fns";

// Inicializa o Firebase Admin SDK.
admin.initializeApp();
const db = admin.firestore();

/**
 * Interface para os dados brutos de estatísticas de um jogador.
 */
interface PlayerStats {
  player_id: string;
  player: string;
  kills: number;
  deaths: number;
  combat: number;
  offense: number;
  defense: number;
  support: number;
  time_seconds: number;
  longest_life_secs: number;
  teamkills: number;
  deaths_by_tk: number;
  weapons: { [key: string]: number };
  death_by: { [key: string]: number };
  most_killed: { [key: string]: number };
}

/**
 * Interface para a estrutura do documento da partida.
 */
interface MatchData {
  id: number;
  numeric_id: number;
  creation_time: string;
  server: string;
  player_stats: PlayerStats[];
  historical?: boolean;
}

/**
 * Retorna o identificador da semana no formato "YYYY-WW" para uma data.
 * @param {Date} date A data para a qual o ID da semana será gerado.
 * @return {string} O identificador da semana.
 */
function getWeekId(date: Date): string {
  // Usa getWeekYear para lidar corretamente com a virada do ano.
  const year = getWeekYear(date, {weekStartsOn: 1});
  // Começa na segunda-feira
  const week = getWeek(date, {weekStartsOn: 1});
  return `week_${year}-${week.toString().padStart(2, "0")}`;
}


/**
 * Retorna o identificador do mês no formato "YYYY-MM" para uma data.
 * @param {Date} date A data para a qual o ID do mês será gerado.
 * @return {string} O identificador do mês.
 */
function getMonthId(date: Date): string {
  const year = date.getFullYear();
  const month = (date.getMonth() + 1).toString().padStart(2, "0");
  return `month_${year}-${month}`;
}

/**
* Gatilho do Firestore acionado na criação de um novo documento de partida.
* Esta função processa dados da partida, agrega estatísticas para cada
* jogador (geral, semanal e mensal) e depois exclui o documento.
*/
export const aggregatePlayerStats = onDocumentCreated(
  {
    document: "rawMatchResults/{matchId}",
    region: "southamerica-east1",
    memory: "512MiB",
    timeoutSeconds: 120,
  },
  async (event) => {
    const {matchId} = event.params;
    const snapshot = event.data;
    if (!snapshot) {
      logger.log(`No data for matchId: ${matchId}. Exiting.`);
      return;
    }
    const matchData = snapshot.data() as MatchData;
    const serverName = matchData.server;
    const matchNumericId = matchData.numeric_id || matchData.id;
    const isHistorical = matchData.historical === true;

    // A data da partida é extraída do documento,
    // com fallback para a data atual.
    const matchDate = matchData.creation_time ?
      new Date(matchData.creation_time) : new Date();

    const logPrefix = `Partida ${matchNumericId} (Servidor: ${serverName})`;
    logger.log(`Iniciando agregação para ${logPrefix}. ` +
      `Data da partida: ${matchDate.toISOString()}. ` +
      `Histórico: ${isHistorical}`);

    if (!matchData.player_stats || matchData.player_stats.length === 0) {
      logger.log(`${logPrefix} não contém 'player_stats'. Excluindo.`);
      await snapshot.ref.delete();
      return;
    }
    if (!serverName) {
      logger.error(`${logPrefix} não tem 'serverName'. Excluindo.`);
      await snapshot.ref.delete();
      return;
    }

    try {
      const playerPromises = matchData.player_stats.map(async (playerStat) => {
        if (!playerStat.player_id) {
          logger.warn("Jogador sem 'player_id' encontrado:", playerStat);
          return Promise.resolve();
        }

        const playerId = playerStat.player_id;
        const playerName = playerStat.player;

        // --- TRANSAÇÃO PARA ESTATÍSTICAS GERAIS ---
        await db.runTransaction(async (transaction) => {
          const aggregatesRef = db.collection("playerAggregates")
            .doc(playerId);
          const playerDoc = await transaction.get(aggregatesRef);
          const currentData = playerDoc.data() || {};
          const currentProcessed = currentData.processedServers || {};

          if ((currentProcessed[serverName] || 0) >= matchNumericId) {
            logger.log(
              `Partida ${matchNumericId} já processada para ${playerId}.`
            );
            return;
          }

          const newLongestLife = Math.max(
            currentData.longestLifeSecs || 0,
            playerStat.longest_life_secs || 0,
          );

          const updates = {
            matchesPlayed: admin.firestore.FieldValue.increment(1),
            totalKills: admin.firestore.FieldValue.increment(
              playerStat.kills || 0
            ),
            totalDeaths: admin.firestore.FieldValue.increment(
              playerStat.deaths || 0
            ),
            totalCombat: admin.firestore.FieldValue.increment(
              playerStat.combat || 0
            ),
            totalOffense: admin.firestore.FieldValue.increment(
              playerStat.offense || 0
            ),
            totalDefense: admin.firestore.FieldValue.increment(
              playerStat.defense || 0
            ),
            totalSupport: admin.firestore.FieldValue.increment(
              playerStat.support || 0
            ),
            totalTimeSeconds: admin.firestore.FieldValue.increment(
              playerStat.time_seconds || 0
            ),
            totalTeamKills: admin.firestore.FieldValue.increment(
              playerStat.teamkills || 0
            ),
            totalDeathsByTK: admin.firestore.FieldValue.increment(
              playerStat.deaths_by_tk || 0
            ),
            longestLifeSecs: newLongestLife,
            playerId: playerId,
            latestPlayerName: playerName,
            [`processedServers.${serverName}`]: matchNumericId,
          };
          transaction.set(aggregatesRef, updates, {merge: true});
        });

        // --- ATUALIZAÇÕES PARA ESTATÍSTICAS SEMANAIS E MENSAIS ---
        if (!isHistorical) {
          const currentDate = new Date();
          const matchWeekId = getWeekId(matchDate);
          const currentWeekId = getWeekId(currentDate);
          const matchMonthId = getMonthId(matchDate);
          const currentMonthId = getMonthId(currentDate);

          const periodBatch = db.batch();
          let periodUpdatesMade = false;

          const periodIncrementData = {
            totalKills: admin.firestore.FieldValue.increment(
              playerStat.kills || 0
            ),
            totalDeaths: admin.firestore.FieldValue.increment(
              playerStat.deaths || 0
            ),
            totalCombat: admin.firestore.FieldValue.increment(
              playerStat.combat || 0
            ),
            totalOffense: admin.firestore.FieldValue.increment(
              playerStat.offense || 0
            ),
            totalDefense: admin.firestore.FieldValue.increment(
              playerStat.defense || 0
            ),
            totalSupport: admin.firestore.FieldValue.increment(
              playerStat.support || 0
            ),
            totalTimeSeconds: admin.firestore.FieldValue.increment(
              playerStat.time_seconds || 0
            ),
            totalTeamKills: admin.firestore.FieldValue.increment(
              playerStat.teamkills || 0
            ),
            totalDeathsByTK: admin.firestore.FieldValue.increment(
              playerStat.deaths_by_tk || 0
            ),
            longestLifeSecs: playerStat.longest_life_secs || 0,
          };
          const basePeriodData = {
            playerId: playerId,
            latestPlayerName: playerName,
          };

          // Agrega para a semana APENAS se a partida for da semana atual
          if (matchWeekId === currentWeekId) {
            const weeklyDocId = `${playerId}_${currentWeekId}`;
            const weeklyRef = db.collection("playerWeeklyStats")
              .doc(weeklyDocId);
            periodBatch.set(weeklyRef, {
              ...basePeriodData, periodId: currentWeekId, ...periodIncrementData,
            }, {merge: true});
            periodUpdatesMade = true;
          }

          // Agrega para o mês APENAS se a partida for do mês atual
          if (matchMonthId === currentMonthId) {
            const monthlyDocId = `${playerId}_${currentMonthId}`;
            const monthlyRef = db.collection("playerMonthlyStats")
              .doc(monthlyDocId);
            periodBatch.set(monthlyRef, {
              ...basePeriodData, periodId: currentMonthId, ...periodIncrementData,
            }, {merge: true});
            periodUpdatesMade = true;
          }

          if (periodUpdatesMade) {
            await periodBatch.commit();
          }
        }
      });

      await Promise.all(playerPromises);

      await snapshot.ref.delete();
      logger.log(`${logPrefix} agregada e documento excluído com sucesso.`);
    } catch (error) {
      logger.error(`Erro ao agregar ${logPrefix}:`, error);
      // Mesmo com erro, exclui para não tentar reprocessar uma partida inválida
      await snapshot.ref.delete();
    }
  }
);
