import {logger} from "firebase-functions";
import {onDocumentCreated} from "firebase-functions/v2/firestore";
import {onSchedule} from "firebase-functions/v2/scheduler";
import * as admin from "firebase-admin";

// Inicializa o Firebase Admin SDK.
admin.initializeApp();
const db = admin.firestore();

/**
 * Interface para os dados brutos de estatísticas de um jogador.
 */
interface PlayerStats {
  player_id: string;
  player: string;
  kills: number;
  deaths: number;
  combat: number;
  offense: number;
  defense: number;
  support: number;
  time_seconds: number;
  longest_life_secs: number;
  teamkills: number;
  deaths_by_tk: number;
  kills_streak?: number;
  vehicle_kills?: number;
  vehicles_destroyed?: number;
  weapons: { [key: string]: number };
  death_by: { [key: string]: number };
  most_killed: { [key: string]: number };
  units?: { ts: number; team: number; squad: number; role: number }[];
}

/**
 * Interface para a estrutura do documento da partida.
 */
interface MatchData {
  id: number;
  numeric_id: number;
  creation_time: string;
  server: string;
  match_time?: number;
  player_stats: PlayerStats[];
  historical?: boolean;
}

/**
 * Retorna o identificador do mês no formato "YYYY-MM" para uma data.
 * @param {Date} date A data para a qual o ID do mês será gerado.
 * @return {string} O identificador do mês.
 */
function getMonthId(date: Date): string {
  const year = date.getFullYear();
  const month = (date.getMonth() + 1).toString().padStart(2, "0");
  return `month_${year}-${month}`;
}

/**
 * Gatilho do Firestore acionado na criação de um novo documento de partida.
 * Esta função processa dados da partida, agrega estatísticas para cada
 * jogador (geral, semanal e mensal) e depois exclui o documento.
 */
export const aggregatePlayerStats = onDocumentCreated(
  {
    document: "rawMatchResults/{matchId}",
    region: "southamerica-east1",
    memory: "512MiB",
    timeoutSeconds: 120,
  },
  async (event) => {
    const {matchId} = event.params;
    const snapshot = event.data;
    if (!snapshot) {
      logger.log(`No data for matchId: ${matchId}. Exiting.`);
      return;
    }
    const matchData = snapshot.data() as MatchData;
    const serverName = matchData.server;
    const matchNumericId = matchData.numeric_id || matchData.id;
    const isHistorical = matchData.historical === true;

    // A data da partida é extraída do documento,
    // com fallback para a data atual.
    const matchDate = matchData.creation_time ?
      new Date(matchData.creation_time) : new Date();

    const logPrefix = `Partida ${matchNumericId} (Servidor: ${serverName})`;
    logger.log(`Iniciando agregação para ${logPrefix}. ` +
      `Data da partida: ${matchDate.toISOString()}. ` +
      `Histórico: ${isHistorical}`);

    if (!matchData.player_stats || matchData.player_stats.length === 0) {
      logger.log(`${logPrefix} não contém 'player_stats'. Excluindo.`);
      await snapshot.ref.delete();
      return;
    }
    if (!serverName) {
      logger.error(`${logPrefix} não tem 'serverName'. Excluindo.`);
      await snapshot.ref.delete();
      return;
    }

    try {
      const playerPromises = matchData.player_stats.map(async (playerStat) => {
        if (!playerStat.player_id) {
          logger.warn("Jogador sem 'player_id' encontrado:", playerStat);
          return Promise.resolve();
        }

        const playerId = playerStat.player_id;
        const playerName = playerStat.player;

        // --- CÁLCULO DAS CLASSES (ROLES) ---
        const timePlayedByRole: Record<number, number> = {};
        const units = playerStat.units || [];
        for (let i = 0; i < units.length; i++) {
          const currentUnit = units[i];
          const nextUnit = units[i + 1];
          const endTs = nextUnit ? nextUnit.ts : (matchData.match_time || 5400);
          const duration = endTs - currentUnit.ts;
          if (duration > 0 && currentUnit.role !== -111) {
            timePlayedByRole[currentUnit.role] =
              (timePlayedByRole[currentUnit.role] || 0) + duration;
          }
        }

        if (!isHistorical) {
          const matchMonthId = getMonthId(matchDate);

          // --- NOVO RANKING MENSAL (monthly_player_stats) ---
          await db.runTransaction(async (transaction) => {
            const newMonthlyDocId = `${matchMonthId}_${playerId}`;
            const newMonthlyRef = db.collection("monthly_player_stats").doc(newMonthlyDocId);
            const docSnap = await transaction.get(newMonthlyRef);
            const currentMonthly = docSnap.data() || {};

            // Mesclar timePlayedByRole
            const mergedRoles = {...(currentMonthly.timePlayedByRole || {})};
            for (const [r, t] of Object.entries(timePlayedByRole)) {
              mergedRoles[Number(r)] = (mergedRoles[Number(r)] || 0) + t;
            }

            // Recalcular mainRole
            let updatedMainRole = -1;
            let updatedMaxTime = -1;
            for (const [r, t] of Object.entries(mergedRoles)) {
              if ((t as number) > updatedMaxTime) {
                updatedMaxTime = t as number;
                updatedMainRole = Number(r);
              }
            }

            // Mesclar armas (topWeapons)
            const mergedWeapons = {...(currentMonthly.topWeapons || {})};
            for (const [weapon, count] of Object.entries(playerStat.weapons || {})) {
              mergedWeapons[weapon] = (mergedWeapons[weapon] || 0) + count;
            }

            // Mesclar vítimas (topVictims)
            const mergedVictims = {...(currentMonthly.topVictims || {})};
            for (const [victim, count] of Object.entries(playerStat.most_killed || {})) {
              mergedVictims[victim] = (mergedVictims[victim] || 0) + count;
            }

            // Mesclar algozes (topKilledBy)
            const mergedAlgozes = {...(currentMonthly.topKilledBy || {})};
            for (const [algoz, count] of Object.entries(playerStat.death_by || {})) {
              mergedAlgozes[algoz] = (mergedAlgozes[algoz] || 0) + count;
            }

            const newMaxStreak = Math.max(currentMonthly.maxKillsStreak || 0, playerStat.kills_streak || 0);
            const newLongestLifeMensal = Math.max(currentMonthly.longestLifeSecs || 0, playerStat.longest_life_secs || 0);

            const increment = admin.firestore.FieldValue.increment;

            const monthlyUpdates = {
              id: newMonthlyDocId,
              playerId: playerId,
              playerName: playerName,
              month: matchMonthId,
              totalKills: increment(playerStat.kills || 0),
              totalDeaths: increment(playerStat.deaths || 0),
              totalCombat: increment(playerStat.combat || 0),
              totalOffense: increment(playerStat.offense || 0),
              totalDefense: increment(playerStat.defense || 0),
              totalSupport: increment(playerStat.support || 0),
              totalTimeSeconds: increment(playerStat.time_seconds || 0),
              totalVehicleKills: increment(playerStat.vehicle_kills || 0),
              totalVehiclesDestroyed: increment(playerStat.vehicles_destroyed || 0),
              totalTeamkills: increment(playerStat.teamkills || 0),
              deathsByTk: increment(playerStat.deaths_by_tk || 0),
              matchesPlayed: increment(1),
              longestLifeSecs: newLongestLifeMensal,
              maxKillsStreak: newMaxStreak,
              timePlayedByRole: mergedRoles,
              mainRole: updatedMainRole,
              topWeapons: mergedWeapons,
              topVictims: mergedVictims,
              topKilledBy: mergedAlgozes,
            };

            transaction.set(newMonthlyRef, monthlyUpdates, {merge: true});
          });
        }
      });

      await Promise.all(playerPromises);

      await snapshot.ref.delete();
      logger.log(`${logPrefix} agregada e documento excluído com sucesso.`);
    } catch (error) {
      logger.error(`Erro ao agregar ${logPrefix}:`, error);
      // Mesmo com erro, exclui para não tentar reprocessar uma partida inválida
      await snapshot.ref.delete();
    }
  }
);

// =================================================================
// NOVA FUNÇÃO DE SINCRONIZAÇÃO AUTOMÁTICA (INTELIGENTE POR DATA)
// =================================================================

const delay = (ms: number) => new Promise((resolve) => setTimeout(resolve, ms));

/**
 * Função agendada para sincronizar automaticamente novas partidas
 * de servidores públicos.
 * Acionada a cada 3 horas. Busca partidas dos últimos 7 dias.
 */
export const autoSyncPublicMatches =
onSchedule("every 3 hours", async () => {
  logger.info("Iniciando rotina de sincronização automática de partidas.");

  // 1. Busca os servidores dinâmicos do banco
  const serversSnap = await db.collection("serversConfig").get();
  const servers = serversSnap.docs.map((doc) => ({
    id: doc.id,
    ...(doc.data() as { name: string; apiUrl: string }),
  }));

  if (servers.length === 0) {
    logger.warn("Nenhum servidor configurado na coleção serversConfig.");
    return;
  }

  // Data limite: 7 dias atrás (sincronização automática foca só nas recentes)
  const cutoffDate = new Date();
  cutoffDate.setDate(cutoffDate.getDate() - 7);

  for (const server of servers) {
    const serverName = server.name;
    const apiUrl = server.apiUrl;
    logger.info(`[SYNC] Sincronizando servidor: ${serverName}`);

    try {
      const syncStatusRef = db.collection("serverSyncStatus").doc(serverName);
      const syncDoc = await syncStatusRef.get();
      let processedIds: number[] = [];
      
      if (syncDoc.exists && syncDoc.data()?.processedIds) {
          processedIds = syncDoc.data()!.processedIds;
      }

      let currentPage = 1;
      let matchesProcessedCount = 0;
      let keepFetching = true;

      while (keepFetching) {
        const url = `${apiUrl}/get_scoreboard_maps?page=${currentPage}&limit=50`;
        const response = await fetch(url);

        if (!response.ok) {
          logger.warn(`[SYNC] Falha API de ${serverName} pág ${currentPage}`);
          break;
        }

        const data = await response.json();
        const maps = data.result?.maps || [];

        if (maps.length === 0) {
            break;
        }

        let allOlder = true;

        for (const m of maps) {
            const matchId = m.id;
            const matchEndDate = new Date(m.end);
            
            if (matchEndDate >= cutoffDate) {
                allOlder = false;
                
                // Trava 1: Duração
                const durationMins = (matchEndDate.getTime() - new Date(m.start).getTime()) / 60000;

                if (!processedIds.includes(matchId)) {
                    if (durationMins < 15) {
                        logger.info(`[SYNC] Partida ${matchId} de ${serverName} descartada por duração curta (${Math.round(durationMins)} min).`);
                        continue;
                    }

                    try {
                        await delay(250);
                        const mapUrl = `${apiUrl}/get_map_scoreboard?map_id=${matchId}`;
                        const mapResponse = await fetch(mapUrl);

                        if (mapResponse.ok) {
                            const mapData = await mapResponse.json();
                            const matchInfo: MatchData = mapData.result;

                            if (matchInfo && matchInfo.player_stats) {
                                
                                // Trava 2: Regra de Seeding Original
                                const totalPlayers = matchInfo.player_stats.length;
                                const totalKills = matchInfo.player_stats.reduce(
                                    (acc, p) => acc + (p.kills || 0), 0
                                );

                                if (totalPlayers < 40 || totalKills < 50) {
                                    logger.info(`[SYNC] Partida ID ${matchId} de ${serverName} ` +
                                    `descartada por seeding (Jogadores: ${totalPlayers}, Kills: ${totalKills}).`);
                                    // Adiciona aos processados para não baixar o JSON completo de novo
                                    processedIds.push(matchId);
                                    continue;
                                }

                                const cleanFirestoreData = (data: any): any => {
                                    if (data === null || typeof data !== 'object') return data;
                                    if (Array.isArray(data)) return data.map(cleanFirestoreData);
                                    const cleaned: any = {};
                                    for (const [key, value] of Object.entries(data)) {
                                        if (key.startsWith('__') && key.endsWith('__')) continue;
                                        cleaned[key] = cleanFirestoreData(value);
                                    }
                                    return cleaned;
                                };

                                const cleanedMatchInfo = cleanFirestoreData(matchInfo);

                                const matchDocRef = db.collection("rawMatchResults").doc(matchInfo.id.toString());
                                await matchDocRef.set({...cleanedMatchInfo, numeric_id: matchInfo.id, server: serverName});

                                processedIds.push(matchId);
                                matchesProcessedCount++;
                            }
                        }
                    } catch (innerError) {
                        logger.error(`[SYNC] Erro processando a partida ${matchId}:`, (innerError as Error).message);
                    }
                }
            }
        }

        if (allOlder) {
            logger.info(`[SYNC] Atingiu limite de 7 dias para ${serverName}. Parando.`);
            keepFetching = false;
        } else {
            currentPage++;
            await delay(200);
        }
      }

      await syncStatusRef.set({
        processedIds: processedIds,
        serverName: serverName,
        lastChecked: admin.firestore.FieldValue.serverTimestamp(),
      }, {merge: true});

      logger.info(`[SYNC] Finalizado ${serverName}. Importadas: ${matchesProcessedCount}.`);

    } catch (error) {
      logger.error(`[SYNC] Erro no servidor ${serverName}:`, (error as Error).message);
    }
  }
});
