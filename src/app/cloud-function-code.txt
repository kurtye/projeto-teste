import {logger} from "firebase-functions";
import {onDocumentCreated} from "firebase-functions/v2/firestore";
import {onSchedule} from "firebase-functions/v2/scheduler";
import * as admin from "firebase-admin";
import {getWeek, getWeekYear} from "date-fns";

// Inicializa o Firebase Admin SDK.
admin.initializeApp();
const db = admin.firestore();

/**
 * Interface para os dados brutos de estatísticas de um jogador.
 */
interface PlayerStats {
  player_id: string;
  player: string;
  kills: number;
  deaths: number;
  combat: number;
  offense: number;
  defense: number;
  support: number;
  time_seconds: number;
  longest_life_secs: number;
  teamkills: number;
  deaths_by_tk: number;
  weapons: { [key: string]: number };
  death_by: { [key: string]: number };
  most_killed: { [key: string]: number };
}

/**
 * Interface para a estrutura do documento da partida.
 */
interface MatchData {
  id: number;
  numeric_id: number;
  creation_time: string;
  server: string;
  player_stats: PlayerStats[];
  historical?: boolean;
}

/**
 * Retorna o identificador da semana no formato "YYYY-WW" para uma data.
 * @param {Date} date A data para a qual o ID da semana será gerado.
 * @return {string} O identificador da semana.
 */
function getWeekId(date: Date): string {
  // Usa getWeekYear para lidar corretamente com a virada do ano.
  const year = getWeekYear(date, {weekStartsOn: 1});
  // Começa na segunda-feira
  const week = getWeek(date, {weekStartsOn: 1});
  return `week_${year}-${week.toString().padStart(2, "0")}`;
}


/**
 * Retorna o identificador do mês no formato "YYYY-MM" para uma data.
 * @param {Date} date A data para a qual o ID do mês será gerado.
 * @return {string} O identificador do mês.
 */
function getMonthId(date: Date): string {
  const year = date.getFullYear();
  const month = (date.getMonth() + 1).toString().padStart(2, "0");
  return `month_${year}-${month}`;
}

/**
* Gatilho do Firestore acionado na criação de um novo documento de partida.
* Esta função processa dados da partida, agrega estatísticas para cada
* jogador (geral, semanal e mensal) e depois exclui o documento.
*/
export const aggregatePlayerStats = onDocumentCreated(
  {
    document: "rawMatchResults/{matchId}",
    region: "southamerica-east1",
    memory: "512MiB",
    timeoutSeconds: 120,
  },
  async (event) => {
    const {matchId} = event.params;
    const snapshot = event.data;
    if (!snapshot) {
      logger.log(`No data for matchId: ${matchId}. Exiting.`);
      return;
    }
    const matchData = snapshot.data() as MatchData;
    const serverName = matchData.server;
    const matchNumericId = matchData.numeric_id || matchData.id;
    const isHistorical = matchData.historical === true;

    // A data da partida é extraída do documento,
    // com fallback para a data atual.
    const matchDate = matchData.creation_time ?
      new Date(matchData.creation_time) : new Date();

    const logPrefix = `Partida ${matchNumericId} (Servidor: ${serverName})`;
    logger.log(`Iniciando agregação para ${logPrefix}. ` +
      `Data da partida: ${matchDate.toISOString()}. ` +
      `Histórico: ${isHistorical}`);

    if (!matchData.player_stats || matchData.player_stats.length === 0) {
      logger.log(`${logPrefix} não contém 'player_stats'. Excluindo.`);
      await snapshot.ref.delete();
      return;
    }
    if (!serverName) {
      logger.error(`${logPrefix} não tem 'serverName'. Excluindo.`);
      await snapshot.ref.delete();
      return;
    }

    try {
      const playerPromises = matchData.player_stats.map(async (playerStat) => {
        if (!playerStat.player_id) {
          logger.warn("Jogador sem 'player_id' encontrado:", playerStat);
          return Promise.resolve();
        }

        const playerId = playerStat.player_id;
        const playerName = playerStat.player;

        // --- TRANSAÇÃO PARA ESTATÍSTICAS GERAIS ---
        await db.runTransaction(async (transaction) => {
          const aggregatesRef = db.collection("playerAggregates")
            .doc(playerId);
          const playerDoc = await transaction.get(aggregatesRef);
          const currentData = playerDoc.data() || {};
          const currentProcessed = currentData.processedServers || {};

          if ((currentProcessed[serverName] || 0) >= matchNumericId) {
            logger.log(
              `Partida ${matchNumericId} já processada para ${playerId}.`
            );
            return;
          }

          const newLongestLife = Math.max(
            currentData.longestLifeSecs || 0,
            playerStat.longest_life_secs || 0,
          );

          const updates = {
            matchesPlayed: admin.firestore.FieldValue.increment(1),
            totalKills: admin.firestore.FieldValue.increment(
              playerStat.kills || 0
            ),
            totalDeaths: admin.firestore.FieldValue.increment(
              playerStat.deaths || 0
            ),
            totalCombat: admin.firestore.FieldValue.increment(
              playerStat.combat || 0
            ),
            totalOffense: admin.firestore.FieldValue.increment(
              playerStat.offense || 0
            ),
            totalDefense: admin.firestore.FieldValue.increment(
              playerStat.defense || 0
            ),
            totalSupport: admin.firestore.FieldValue.increment(
              playerStat.support || 0
            ),
            totalTimeSeconds: admin.firestore.FieldValue.increment(
              playerStat.time_seconds || 0
            ),
            totalTeamKills: admin.firestore.FieldValue.increment(
              playerStat.teamkills || 0
            ),
            totalDeathsByTK: admin.firestore.FieldValue.increment(
              playerStat.deaths_by_tk || 0
            ),
            longestLifeSecs: newLongestLife,
            playerId: playerId,
            latestPlayerName: playerName,
            [`processedServers.${serverName}`]: matchNumericId,
          };
          transaction.set(aggregatesRef, updates, {merge: true});
        });

        // --- BATCH PARA SUBCOLEÇÕES E ESTATÍSTICAS GLOBAIS ---
        const subcollectionBatch = db.batch();
        const aggregatesRef = db.collection("playerAggregates").doc(playerId);

        // Agrega estatísticas de armas (por jogador e global)
        if (playerStat.weapons) {
          const weaponsRef = aggregatesRef.collection("weaponUsage");
          for (const weapon in playerStat.weapons) {
            const kills = playerStat.weapons[weapon];
            if (kills > 0) {
              const weaponDocRef = weaponsRef.doc(weapon);
              subcollectionBatch.set(
                weaponDocRef,
                {name: weapon, count: admin.firestore.FieldValue.increment(kills)},
                {merge: true}
              );
              // Incrementa estatísticas globais de armas
              const globalWeaponRef = db.collection("globalWeaponStats").doc(weapon);
              subcollectionBatch.set(globalWeaponRef, {
                name: weapon,
                totalKills: admin.firestore.FieldValue.increment(kills),
              }, {merge: true});
            }
          }
        }

        // Agrega estatísticas de 'morto por'
        if (playerStat.death_by) {
            const killedByRef = aggregatesRef.collection("killedBy");
            for (const killer in playerStat.death_by) {
                const times = playerStat.death_by[killer];
                if (times > 0) {
                    const killerDocRef = killedByRef.doc(killer);
                    subcollectionBatch.set(
                        killerDocRef,
                        {name: killer, count: admin.firestore.FieldValue.increment(times)},
                        {merge: true}
                    );
                }
            }
        }

        // Agrega estatísticas de 'mais matou'
        if (playerStat.most_killed) {
            const killedPlayersRef = aggregatesRef.collection("killedPlayers");
            for (const victim in playerStat.most_killed) {
                const times = playerStat.most_killed[victim];
                if (times > 0) {
                    const victimDocRef = killedPlayersRef.doc(victim);
                    subcollectionBatch.set(
                        victimDocRef,
                        {name: victim, count: admin.firestore.FieldValue.increment(times)},
                        {merge: true}
                    );
                }
            }
        }
        await subcollectionBatch.commit();


        // --- ATUALIZAÇÕES PARA ESTATÍSTICAS SEMANAIS E MENSAIS ---
        if (!isHistorical) {
          const currentDate = new Date();
          const matchWeekId = getWeekId(matchDate);
          const currentWeekId = getWeekId(currentDate);
          const matchMonthId = getMonthId(matchDate);
          const currentMonthId = getMonthId(currentDate);

          const periodBatch = db.batch();
          let periodUpdatesMade = false;

          const periodIncrementData = {
            totalKills: admin.firestore.FieldValue.increment(
              playerStat.kills || 0
            ),
            totalDeaths: admin.firestore.FieldValue.increment(
              playerStat.deaths || 0
            ),
            totalCombat: admin.firestore.FieldValue.increment(
              playerStat.combat || 0
            ),
            totalOffense: admin.firestore.FieldValue.increment(
              playerStat.offense || 0
            ),
            totalDefense: admin.firestore.FieldValue.increment(
              playerStat.defense || 0
            ),
            totalSupport: admin.firestore.FieldValue.increment(
              playerStat.support || 0
            ),
            totalTimeSeconds: admin.firestore.FieldValue.increment(
              playerStat.time_seconds || 0
            ),
            totalTeamKills: admin.firestore.FieldValue.increment(
              playerStat.teamkills || 0
            ),
            totalDeathsByTK: admin.firestore.FieldValue.increment(
              playerStat.deaths_by_tk || 0
            ),
            longestLifeSecs: playerStat.longest_life_secs || 0,
          };
          const basePeriodData = {
            playerId: playerId,
            latestPlayerName: playerName,
          };

          // Agrega para a semana APENAS se a partida for da semana atual
          if (matchWeekId === currentWeekId) {
            const weeklyDocId = `${playerId}_${currentWeekId}`;
            const weeklyRef = db.collection("playerWeeklyStats")
              .doc(weeklyDocId);
            periodBatch.set(weeklyRef, {
              ...basePeriodData, periodId: currentWeekId, ...periodIncrementData,
            }, {merge: true});
            periodUpdatesMade = true;
          }

          // Agrega para o mês APENAS se a partida for do mês atual
          if (matchMonthId === currentMonthId) {
            const monthlyDocId = `${playerId}_${currentMonthId}`;
            const monthlyRef = db.collection("playerMonthlyStats")
              .doc(monthlyDocId);
            periodBatch.set(monthlyRef, {
              ...basePeriodData, periodId: currentMonthId, ...periodIncrementData,
            }, {merge: true});
            periodUpdatesMade = true;
          }

          if (periodUpdatesMade) {
            await periodBatch.commit();
          }
        }
      });

      await Promise.all(playerPromises);

      await snapshot.ref.delete();
      logger.log(`${logPrefix} agregada e documento excluído com sucesso.`);
    } catch (error) {
      logger.error(`Erro ao agregar ${logPrefix}:`, error);
      // Mesmo com erro, exclui para não tentar reprocessar uma partida inválida
      await snapshot.ref.delete();
    }
  }
);

const serversConfig = [
  {id: "3LPZ", name: "3LPZ", apiUrl: "https://3lpz-stats.hlladmin.com/api"},
  {id: "HRB", name: "HRB", apiUrl: "https://hrb-stats.hlladmin.com/api"},
  {id: "RZN", name: "RZN", apiUrl: "https://rzn-stats.crcon.cc/api"},
  {id: "GOAT", name: "GOAT", apiUrl: "https://goat-stats.hlladmin.com/api"},
  {id: "OCL", name: "OCL", apiUrl: "https://ocabala-stats.hlladmin.com/api"},
  {id: "SAP", name: "SAP", apiUrl: "https://sap-stats.hlladmin.com/api"},
  {id: "SOH", name: "SOH", apiUrl: "https://sohhllbr-stats.hlladmin.com/api"},
];

const delay = (ms: number) => new Promise((resolve) => setTimeout(resolve, ms));

interface ScoreboardMapsResponse {
  result: {
    total: number;
    page_size: number;
    maps: { id: number }[];
  };
}

// Reutilizando interface de MatchData e PlayerStats de cima

async function fetchAllMatchIds(apiUrl: string): Promise<number[]> {
  let allIds: number[] = [];
  let currentPage = 1;
  let totalPages = 1;

  logger.log(`[SYNC] Buscando todos os IDs de partida de ${apiUrl}`);

  try {
    do {
      const url = `${apiUrl}/get_scoreboard_maps?page=${currentPage}&limit=100`;
      // A API de fetch é global no ambiente do Node.js 18+ do Cloud Functions
      const response = await fetch(url);

      if (!response.ok) {
        const errorText = await response.text();
        logger.error(`[SYNC] Falha ao buscar a página ${currentPage} de mapas. Status: ${response.status}, Corpo: ${errorText}`);
        throw new Error(`Falha ao buscar a lista de mapas na página ${currentPage}: ${response.statusText}`);
      }

      const data: ScoreboardMapsResponse = await response.json();

      if (currentPage === 1) {
        const totalMatches = data.result.total;
        const pageSize = data.result.page_size;
        totalPages = Math.ceil(totalMatches / pageSize);
        logger.log(`[SYNC] Total de partidas na API: ${totalMatches}. Total de páginas a buscar: ${totalPages}.`);
      }

      const pageIds = data.result.maps.map((m) => m.id);
      allIds = allIds.concat(pageIds);

      currentPage++;
      if (currentPage <= totalPages) {
        await delay(200);
      }
    } while (currentPage <= totalPages);

    logger.log(`[SYNC] Busca de IDs concluída. Total de IDs encontrados: ${allIds.length}`);
    return allIds;
  } catch (error) {
    logger.error(`[SYNC] Erro ao buscar todos os IDs de partida de ${apiUrl}:`, error);
    throw error;
  }
}

/**
 * Função agendada para sincronizar automaticamente novas partidas de servidores públicos.
 * Acionada a cada 3 horas.
 */
export const autoSyncPublicMatches = onSchedule("every 3 hours", async (event) => {
  logger.info("Iniciando rotina de sincronização automática de partidas.");

  for (const server of serversConfig) {
    logger.info(`[SYNC] Sincronizando servidor: ${server.name}`);
    try {
      const syncStatusRef = db.collection("serverSyncStatus").doc(server.name);
      const syncDoc = await syncStatusRef.get();
      const lastProcessedId = syncDoc.exists ? (syncDoc.data()?.lastProcessedId || 0) : 0;
      logger.info(`[SYNC] Último ID processado para ${server.name}: ${lastProcessedId}`);

      const allMatchIds = await fetchAllMatchIds(server.apiUrl);
      if (allMatchIds.length === 0) {
        logger.warn(`[SYNC] Nenhum ID de partida encontrado para ${server.name}. Pulando.`);
        continue;
      }

      const matchIdsToImport = allMatchIds.filter((id) => id > lastProcessedId).sort((a, b) => a - b);
      if (matchIdsToImport.length === 0) {
        logger.info(`[SYNC] Nenhuma partida nova para importar de ${server.name}.`);
        continue;
      }

      logger.info(`[SYNC] Encontradas ${matchIdsToImport.length} novas partidas para importar de ${server.name}.`);
      let matchesProcessedCount = 0;
      let highestIdInBatch = 0;

      for (const matchId of matchIdsToImport) {
        try {
          await delay(250);
          const mapUrl = `${server.apiUrl}/get_map_scoreboard?map_id=${matchId}`;
          const response = await fetch(mapUrl);

          if (!response.ok) {
            logger.warn(`[SYNC] Falha ao buscar partida ID ${matchId} de ${server.name}. Status: ${response.status}. Pulando.`);
            continue;
          }

          const mapData = await response.json();
          const matchInfo: MatchData = mapData.result;

          if (!matchInfo || !matchInfo.player_stats) {
            logger.warn(`[SYNC] Dados da partida ID ${matchId} de ${server.name} incompletos. Pulando.`);
            continue;
          }

          const matchDocRef = db.collection("rawMatchResults").doc(matchInfo.id.toString());
          await matchDocRef.set({...matchInfo, numeric_id: matchInfo.id, server: server.name});

          matchesProcessedCount++;
          highestIdInBatch = Math.max(highestIdInBatch, matchId);
        } catch (innerError: any) {
          logger.error(`[SYNC] Erro processando partida ID ${matchId} de ${server.name}:`, innerError.message);
        }
      }

      if (highestIdInBatch > lastProcessedId) {
        await syncStatusRef.set({
          lastProcessedId: highestIdInBatch,
          serverName: server.name,
          lastChecked: admin.firestore.FieldValue.serverTimestamp(),
        }, {merge: true});
        logger.info(`[SYNC] Status de sincronização para ${server.name} atualizado para ID: ${highestIdInBatch}. Total processado no lote: ${matchesProcessedCount}.`);
      }
    } catch (error: any) {
      logger.error(`[ERRO GERAL SYNC] Falha na sincronização do servidor ${server.name}:`, error);
    }
  }
  logger.info("Rotina de sincronização automática concluída.");
});
