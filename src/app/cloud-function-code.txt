import * as functions from "firebase-functions";
import * as admin from "firebase-admin";

// Inicializa o Firebase Admin SDK para ter acesso ao Firestore no back-end.
admin.initializeApp();
const db = admin.firestore();

/**
 * Interface para os dados brutos de estatísticas de um jogador em uma partida.
 * Tipar os dados ajuda a evitar erros.
 */
interface PlayerStats {
  steam_id_64: string; // O ID correto é 'steam_id_64', não 'player_id'
  name: string;
  kills: number;
  deaths: number;
  combat: number;
  offense: number;
  defense: number;
  support: number;
  time_seconds: number;
  longest_life_secs: number;
  teamkills: number;
  deaths_by_tk: number;
  weapons: { [key: string]: number };
  death_by: { [key: string]: number };
  most_killed: { [key: string]: number };
}

/**
 * Interface para a estrutura do documento da partida em `rawMatchResults`.
 */
interface MatchData {
  id: number;
  player_stats: PlayerStats[];
  // Adicione outros campos da partida se precisar deles na função.
}

/**
 * Gatilho do Firestore que é acionado na criação de um novo documento de partida
 * na coleção `rawMatchResults`.
 *
 * Esta função processa os dados da partida, agrega as estatísticas para cada jogador
 * e, ao final, exclui o documento da partida bruta para evitar reprocessamento.
 */
exports.aggregatePlayerStats = functions
  .region("southamerica-east1")
  .runWith({ memory: "512MB", timeoutSeconds: 60 })
  .firestore.document("rawMatchResults/{matchId}")
  .onCreate(async (snapshot, context) => {
    const { matchId } = context.params;
    const matchData = snapshot.data() as MatchData;

    functions.logger.log(`Iniciando agregação para a partida ID: ${matchId}`);

    if (!matchData.player_stats || matchData.player_stats.length === 0) {
      functions.logger.log(`Partida ${matchId} não contém 'player_stats'. Excluindo documento.`);
      await snapshot.ref.delete();
      return;
    }

    const batch = db.batch();

    // Itera sobre cada jogador na partida para preparar os incrementos
    for (const playerStat of matchData.player_stats) {
      if (!playerStat.steam_id_64) {
        functions.logger.warn("Encontrado um jogador sem 'steam_id_64' na partida:", playerStat);
        continue; // Pula para o próximo jogador
      }

      const playerAggregatesRef = db.collection("playerAggregates").doc(playerStat.steam_id_64);

      // --- Somas Acumulativas ---
      batch.set(playerAggregatesRef, {
        totalKills: admin.firestore.FieldValue.increment(playerStat.kills || 0),
        totalDeaths: admin.firestore.FieldValue.increment(playerStat.deaths || 0),
        totalCombat: admin.firestore.FieldValue.increment(playerStat.combat || 0),
        totalOffense: admin.firestore.FieldValue.increment(playerStat.offense || 0),
        totalDefense: admin.firestore.FieldValue.increment(playerStat.defense || 0),
        totalSupport: admin.firestore.FieldValue.increment(playerStat.support || 0),
        totalTimeSeconds: admin.firestore.FieldValue.increment(playerStat.time_seconds || 0),
        totalTeamKills: admin.firestore.FieldValue.increment(playerStat.teamkills || 0),
        totalDeathsByTK: admin.firestore.FieldValue.increment(playerStat.deaths_by_tk || 0),
      }, { merge: true });

      // --- Atualizações de Mapa (Armas, Mortes, etc.) ---
      const weaponUpdates: { [key: string]: admin.firestore.FieldValue } = {};
      if (playerStat.weapons) {
        for (const [weapon, count] of Object.entries(playerStat.weapons)) {
          weaponUpdates[`weaponUsage.${weapon}`] = admin.firestore.FieldValue.increment(count);
        }
      }

      const mostKilledByUpdates: { [key: string]: admin.firestore.FieldValue } = {};
      if (playerStat.death_by) {
        for (const [killer, count] of Object.entries(playerStat.death_by)) {
          mostKilledByUpdates[`mostKilledBy.${killer}`] = admin.firestore.FieldValue.increment(count);
        }
      }

      const mostKilledPlayersUpdates: { [key: string]: admin.firestore.FieldValue } = {};
      if (playerStat.most_killed) {
        for (const [victim, count] of Object.entries(playerStat.most_killed)) {
          mostKilledPlayersUpdates[`mostKilledPlayers.${victim}`] = admin.firestore.FieldValue.increment(count);
        }
      }

      // Aplica as atualizações de mapa no batch
      batch.set(playerAggregatesRef, {
        ...weaponUpdates,
        ...mostKilledByUpdates,
        ...mostKilledPlayersUpdates,
      }, { merge: true });
    }

    try {
        // Commita o batch principal com todos os incrementos atômicos.
        await batch.commit();

        // A lógica para `longestLifeSecs` e atualizações de metadados precisa de transação,
        // pois depende da leitura do valor atual. Executaremos isso em transações separadas
        // por jogador para evitar contenção excessiva.
        await Promise.all(matchData.player_stats.map(async (playerStat) => {
            if (!playerStat.steam_id_64) return;
            const playerAggregatesRef = db.collection("playerAggregates").doc(playerStat.steam_id_64);

            await db.runTransaction(async (transaction) => {
                const playerDoc = await transaction.get(playerAggregatesRef);
                const currentData = playerDoc.data() || {};

                // Regra do MÁXIMO para longestLifeSecs
                const newLongestLife = Math.max(currentData.longestLifeSecs || 0, playerStat.longest_life_secs || 0);

                // Atualização simples para campos de metadados
                transaction.set(playerAggregatesRef, {
                    playerId: playerStat.steam_id_64, // Chave
                    latestPlayerName: playerStat.name, // Último nome encontrado
                    lastProcessedMatchId: matchData.id,
                    longestLifeSecs: newLongestLife,
                }, { merge: true });
            });
        }));

        // Após tudo ser processado com sucesso, exclui o documento da fila.
        await snapshot.ref.delete();
        functions.logger.log(`Agregação da partida ${matchId} concluída com sucesso. Documento excluído.`);

    } catch (error) {
      functions.logger.error(`Erro ao agregar estatísticas para a partida ${matchId}:`, error);
      // Não exclui o documento para que ele possa ser reprocessado ou inspecionado.
    }
});
