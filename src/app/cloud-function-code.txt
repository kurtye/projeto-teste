import {logger} from "firebase-functions";
import {onDocumentCreated} from "firebase-functions/v2/firestore";
import * as admin from "firebase-admin";

// Inicializa o Firebase Admin SDK para ter acesso ao Firestore no back-end.
admin.initializeApp();
const db = admin.firestore();

/**
 * Interface para os dados brutos de estatísticas de um jogador em uma partida.
 */
interface PlayerStats {
    player_id: string;
    player: string;
    kills: number;
    deaths: number;
    combat: number;
    offense: number;
    defense: number;
    support: number;
    time_seconds: number;
    longest_life_secs: number;
    teamkills: number;
    deaths_by_tk: number;
    weapons: { [key: string]: number };
    death_by: { [key: string]: number };
    most_killed: { [key: string]: number };
}

/**
 * Interface para a estrutura do documento da partida em `rawMatchResults`.
 */
interface MatchData {
    id: number;
    numeric_id: number;
    server: string;
    player_stats: PlayerStats[];
}

/**
* Gatilho do Firestore que é acionado na criação de um novo documento de partida.
* Esta função processa os dados da partida, agrega as estatísticas para cada jogador
* e, ao final, exclui o documento para evitar reprocessamento.
*/
export const aggregatePlayerStats = onDocumentCreated(
  {
    document: "rawMatchResults/{matchId}",
    region: "southamerica-east1",
    memory: "512MiB",
    timeoutSeconds: 60,
  },
  async (event) => {
    const {matchId} = event.params;
    const snapshot = event.data;
    if (!snapshot) {
      logger.log(`No data associated with event for matchId: ${matchId}. Exiting.`);
      return;
    }
    const matchData = snapshot.data() as MatchData;
    const serverName = matchData.server;
    const matchNumericId = matchData.numeric_id || matchData.id;

    logger.log(`Iniciando agregação para a partida ID: ${matchNumericId} do servidor: ${serverName}`);

    if (!matchData.player_stats || matchData.player_stats.length === 0) {
      logger.log(`Partida ${matchNumericId} não contém 'player_stats'. Excluindo documento.`);
      await snapshot.ref.delete();
      return;
    }
    
    if (!serverName) {
        logger.error(`Partida ${matchNumericId} não tem 'serverName'. Impossível processar. Documento será excluído para evitar loops.`);
        await snapshot.ref.delete();
        return;
    }

    try {
      const playerUpdatePromises = matchData.player_stats.map((playerStat) => {
        if (!playerStat.player_id) {
          logger.warn("Encontrado um jogador sem 'player_id' na partida, pulando:", playerStat);
          return Promise.resolve();
        }

        const playerAggregatesRef = db.collection("playerAggregates").doc(playerStat.player_id);

        return db.runTransaction(async (transaction) => {
          const playerDoc = await transaction.get(playerAggregatesRef);
          const currentData = playerDoc.data() || {};
          const currentProcessedServers = currentData.processedServers || {};

          // VERIFICAÇÃO DE IDEMPOTÊNCIA POR SERVIDOR
          if ((currentProcessedServers[serverName] || 0) >= matchNumericId) {
            logger.log(`Partida ${matchNumericId} do servidor ${serverName} já processada para o jogador ${playerStat.player_id}. Pulando.`);
            return;
          }

          const newLongestLife = Math.max(currentData.longestLifeSecs || 0, playerStat.longest_life_secs || 0);

          const updates: { [key: string]: any } = {
            totalKills: admin.firestore.FieldValue.increment(playerStat.kills || 0),
            totalDeaths: admin.firestore.FieldValue.increment(playerStat.deaths || 0),
            totalCombat: admin.firestore.FieldValue.increment(playerStat.combat || 0),
            totalOffense: admin.firestore.FieldValue.increment(playerStat.offense || 0),
            totalDefense: admin.firestore.FieldValue.increment(playerStat.defense || 0),
            totalSupport: admin.firestore.FieldValue.increment(playerStat.support || 0),
            totalTimeSeconds: admin.firestore.FieldValue.increment(playerStat.time_seconds || 0),
            totalTeamKills: admin.firestore.FieldValue.increment(playerStat.teamkills || 0),
            totalDeathsByTK: admin.firestore.FieldValue.increment(playerStat.deaths_by_tk || 0),
            longestLifeSecs: newLongestLife,
            playerId: playerStat.player_id,
            latestPlayerName: playerStat.player,
            // ATUALIZA O MAPA COM O NOVO ID PARA O SERVIDOR ESPECÍFICO
            [`processedServers.${serverName}`]: matchNumericId,
          };

          if (playerStat.weapons) {
            for (const [weapon, count] of Object.entries(playerStat.weapons)) {
              updates[`weaponUsage.${weapon}`] = admin.firestore.FieldValue.increment(count);
            }
          }
          if (playerStat.death_by) {
            for (const [killer, count] of Object.entries(playerStat.death_by)) {
              updates[`mostKilledBy.${killer}`] = admin.firestore.FieldValue.increment(count);
            }
          }
          if (playerStat.most_killed) {
            for (const [victim, count] of Object.entries(playerStat.most_killed)) {
              updates[`mostKilledPlayers.${victim}`] = admin.firestore.FieldValue.increment(count);
            }
          }
          transaction.set(playerAggregatesRef, updates, {merge: true});
        });
      });

      await Promise.all(playerUpdatePromises);
      await snapshot.ref.delete();
      logger.log(`Partida ${matchNumericId} do servidor ${serverName} agregada e documento excluído com sucesso.`);
    } catch (error) {
      logger.error(`Erro ao agregar estatísticas para a partida ${matchNumericId}:`, error);
    }
  }
);
